# llm-proxy backend core — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Working LLM gateway: users authenticate locally or via OIDC, issue API tokens, and every proxied request is checked against the owner's model policy and recorded for metrics.

**Architecture:** One Go process embeds CLIProxyAPI v7 as a library. Our gin middleware resolves the token, evaluates the policy predicate and filters `/v1/models`; an in-process usage sink writes the consumption ledger. PostgreSQL is the source of truth — including the upstream configuration, which is pushed as a struct, never a file.

**Tech Stack:** Go 1.26, `github.com/router-for-me/CLIProxyAPI/v7 v7.3.12`, gin, pgx/v5, goose, prometheus/client_golang, coreos/go-oidc, testcontainers-go, mockery.

**Spec:** `docs/specs/2026-09-22-llm-proxy-design.md`

## Global Constraints

- Module path: `github.com/elleqt/llm-proxy-backend`.
- Upstream pinned at `github.com/router-for-me/CLIProxyAPI/v7 v7.3.12`. Upstream `docs/sdk-*.md` are stale (they say `/v6`); verify against source in the sibling `upstream/` clone.
- Never import `github.com/router-for-me/CLIProxyAPI/v7/internal/...` except through the `sdk/config` type aliases.
- `/v0/management` must never be enabled: leave `RemoteManagement.SecretKey` empty, pass no local management password, set no management env secret.
- No hostname, domain, realm, group name, IP or deployment-specific path in code, tests, comments or examples. Use `example.com` and generic names.
- Secrets never appear in API responses or logs, except the single display at creation.
- All mocks are generated by mockery. Hand-written mocks are prohibited; the one exception is the behavioural fake provider executor (Task 10).
- Every behaviour is covered by an automated test. Verifying a feature by hand with curl is not acceptance.
- Restrictions bind to the user, never to the token.

// internal/config/config.go
package config

import (
	"errors"
	"os"
)

type Config struct {
	ListenAddr  string
	DatabaseURL string
	RuntimeDir  string
	AuthDir     string
}

func Load() (Config, error) {
	cfg := Config{
		ListenAddr:  envOr("LLMPROXY_LISTEN_ADDR", ":8080"),
		DatabaseURL: os.Getenv("LLMPROXY_DATABASE_URL"),
		RuntimeDir:  envOr("LLMPROXY_RUNTIME_DIR", "/var/lib/llmproxy/runtime"),
		AuthDir:     envOr("LLMPROXY_AUTH_DIR", "/var/lib/llmproxy/auths"),
	}
	if cfg.DatabaseURL == "" {
		return Config{}, errors.New("config: LLMPROXY_DATABASE_URL is required")
	}
	return cfg, nil
}

func envOr(key, fallback string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return fallback
}

package postgres

import (
	"context"
	"database/sql"
	"embed"
	"fmt"

	"github.com/pressly/goose/v3"
	_ "github.com/jackc/pgx/v5/stdlib"
)

//go:embed migrations/*.sql
var migrations embed.FS

func Migrate(ctx context.Context, dsn string) error {
	db, err := sql.Open("pgx", dsn)
	if err != nil {
		return fmt.Errorf("postgres: open: %w", err)
	}
	defer func() { _ = db.Close() }()

	goose.SetBaseFS(migrations)
	if err := goose.SetDialect("postgres"); err != nil {
		return fmt.Errorf("postgres: dialect: %w", err)
	}
	if err := goose.UpContext(ctx, db, "migrations"); err != nil {
		return fmt.Errorf("postgres: migrate: %w", err)
	}
	return nil
}

```go
// internal/domain/access/policy.go
package access

import (
	"errors"
	"fmt"
	"path"
	"strings"
)

// Rule allows one provider and the models matching ModelPattern.
// ModelPattern uses shell glob syntax; "*" means every model of the provider,
// including models published after the rule was written.
type Rule struct {
	Provider     string
	ModelPattern string
}

// Policy is an allow-list. Order is irrelevant: a request is permitted when any
// rule matches. There are deliberately no deny rules.
type Policy []Rule

var ErrMalformedRule = errors.New("access: malformed rule")

func ParseRule(s string) (Rule, error) {
	parts := strings.Split(strings.TrimSpace(s), ":")
	if len(parts) != 2 {
		return Rule{}, fmt.Errorf("%w: %q", ErrMalformedRule, s)
	}
	provider, pattern := strings.TrimSpace(parts[0]), strings.TrimSpace(parts[1])
	if provider == "" || pattern == "" {
		return Rule{}, fmt.Errorf("%w: %q", ErrMalformedRule, s)
	}
	return Rule{Provider: strings.ToLower(provider), ModelPattern: strings.ToLower(pattern)}, nil
}

func (r Rule) String() string { return r.Provider + ":" + r.ModelPattern }

func (r Rule) matches(provider, model string) bool {
	if r.Provider != "*" && r.Provider != strings.ToLower(provider) {
		return false
	}
	if r.ModelPattern == "*" {
		return true
	}
	ok, err := path.Match(r.ModelPattern, strings.ToLower(model))
	return err == nil && ok
}

func (p Policy) Allows(provider, model string) bool {
	for _, r := range p {
		if r.matches(provider, model) {
			return true
		}
	}
	return false
}
```

// internal/domain/credentials/token.go
package credentials

import (
	"crypto/rand"
	"crypto/sha256"
	"encoding/base64"
	"encoding/hex"
	"errors"
	"fmt"
	"time"

	"github.com/google/uuid"
)

const secretBytes = 32

var ErrAlreadyRevoked = errors.New("credentials: token already revoked")

type Token struct {
	ID         uuid.UUID
	UserID     uuid.UUID
	Label      string
	Hash       string
	Prefix     string
	CreatedAt  time.Time
	LastUsedAt *time.Time
	RevokedAt  *time.Time
	RevokedBy  *uuid.UUID
}

// Generate returns the record to store and the secret to show exactly once.
// The secret is never recoverable from the record.
func Generate(owner uuid.UUID, label string) (Token, string, error) {
	raw := make([]byte, secretBytes)
	if _, err := rand.Read(raw); err != nil {
		return Token{}, "", fmt.Errorf("credentials: entropy: %w", err)
	}
	secret := "sk-" + base64.RawURLEncoding.EncodeToString(raw)
	return Token{
		ID:        uuid.New(),
		UserID:    owner,
		Label:     label,
		Hash:      HashSecret(secret),
		Prefix:    secret[:7],
		CreatedAt: time.Now().UTC(),
	}, secret, nil
}

func HashSecret(secret string) string {
	sum := sha256.Sum256([]byte(secret))
	return hex.EncodeToString(sum[:])
}

func (t Token) Active() bool { return t.RevokedAt == nil }

func (t *Token) Revoke(by uuid.UUID, at time.Time) error {
	if t.RevokedAt != nil {
		return ErrAlreadyRevoked
	}
	when := at.UTC()
	t.RevokedAt = &when
	t.RevokedBy = &by
	return nil
}
```

// internal/infra/postgres/tokens.go
package postgres

import (
	"context"
	"errors"
	"time"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/elleqt/llm-proxy-backend/internal/app"
	"github.com/elleqt/llm-proxy-backend/internal/domain/credentials"
)

type TokenRepo struct{ pool *pgxpool.Pool }

func NewTokenRepo(pool *pgxpool.Pool) *TokenRepo { return &TokenRepo{pool: pool} }

func (r *TokenRepo) Create(ctx context.Context, t credentials.Token) error {
	_, err := r.pool.Exec(ctx,
		`INSERT INTO api_tokens (id, user_id, label, hash, prefix, created_at)
		 VALUES ($1, $2, $3, $4, $5, $6)`,
		t.ID, t.UserID, t.Label, t.Hash, t.Prefix, t.CreatedAt)
	return err
}

func (r *TokenRepo) ByHash(ctx context.Context, hash string) (credentials.Token, error) {
	var t credentials.Token
	err := r.pool.QueryRow(ctx,
		`SELECT id, user_id, label, hash, prefix, created_at, last_used_at, revoked_at, revoked_by
		 FROM api_tokens WHERE hash = $1`, hash).
		Scan(&t.ID, &t.UserID, &t.Label, &t.Hash, &t.Prefix, &t.CreatedAt,
			&t.LastUsedAt, &t.RevokedAt, &t.RevokedBy)
	if errors.Is(err, pgx.ErrNoRows) {
		return credentials.Token{}, app.ErrNotFound
	}
	return t, err
}

func (r *TokenRepo) ListByUser(ctx context.Context, userID uuid.UUID) ([]credentials.Token, error) {
	rows, err := r.pool.Query(ctx,
		`SELECT id, user_id, label, hash, prefix, created_at, last_used_at, revoked_at, revoked_by
		 FROM api_tokens WHERE user_id = $1 ORDER BY created_at DESC`, userID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var out []credentials.Token
	for rows.Next() {
		var t credentials.Token
		if err := rows.Scan(&t.ID, &t.UserID, &t.Label, &t.Hash, &t.Prefix, &t.CreatedAt,
			&t.LastUsedAt, &t.RevokedAt, &t.RevokedBy); err != nil {
			return nil, err
		}
		out = append(out, t)
	}
	return out, rows.Err()
}

func (r *TokenRepo) Save(ctx context.Context, t credentials.Token) error {
	_, err := r.pool.Exec(ctx,
		`UPDATE api_tokens SET label = $2, revoked_at = $3, revoked_by = $4 WHERE id = $1`,
		t.ID, t.Label, t.RevokedAt, t.RevokedBy)
	return err
}

func (r *TokenRepo) TouchLastUsed(ctx context.Context, id uuid.UUID, at time.Time) error {
	_, err := r.pool.Exec(ctx,
		`UPDATE api_tokens SET last_used_at = $2 WHERE id = $1`, id, at.UTC())
	return err
}
```

`access.ParseRule`, on write through `(Rule).String()`. Every `pgx.ErrNoRows` becomes

The watcher factory captures `reload` and returns a nil wrapper; every `s.watcher` call site upstream is nil-guarded, so nothing dereferences it.

```go
// internal/infra/gateway/service.go
package gateway

import (
	"context"
	"errors"
	"sync"

	"github.com/router-for-me/CLIProxyAPI/v7/sdk/cliproxy"
	cliproxyconfig "github.com/router-for-me/CLIProxyAPI/v7/sdk/config"
	sdkapi "github.com/router-for-me/CLIProxyAPI/v7/sdk/api"
	"github.com/gin-gonic/gin"
)

type Params struct {
	Config     *cliproxyconfig.Config
	ConfigPath string // required by Build; never read because our watcher owns config
	Middleware []gin.HandlerFunc
	UsagePlugin usagePlugin
}

type Gateway struct {
	svc *cliproxy.Service

	mu      sync.RWMutex
	current *cliproxyconfig.Config
	reload  func(*cliproxyconfig.Config)
}

var ErrNotRunning = errors.New("gateway: reload callback not captured yet")

func New(p Params) (*Gateway, error) {
	g := &Gateway{current: p.Config}

	builder := cliproxy.NewBuilder().
		WithConfig(p.Config).
		WithConfigPath(p.ConfigPath).
		WithWatcherFactory(func(_, _ string, reload func(*cliproxyconfig.Config)) (*cliproxy.WatcherWrapper, error) {
			g.mu.Lock()
			g.reload = reload
			g.mu.Unlock()
			return nil, nil // hollow watcher: no file, no fsnotify
		})
	if len(p.Middleware) > 0 {
		builder = builder.WithServerOptions(sdkapi.WithMiddleware(p.Middleware...))
	}

	svc, err := builder.Build()
	if err != nil {
		return nil, err
	}
	if p.UsagePlugin != nil {
		svc.RegisterUsagePlugin(p.UsagePlugin)
	}
	g.svc = svc
	return g, nil
}

func (g *Gateway) Run(ctx context.Context) error { return g.svc.Run(ctx) }

// PushConfig applies a configuration built from the database.
func (g *Gateway) PushConfig(cfg *cliproxyconfig.Config) error {
	g.mu.RLock()
	reload := g.reload
	g.mu.RUnlock()
	if reload == nil {
		return ErrNotRunning
	}
	reload(cfg)
	g.mu.Lock()
	g.current = cfg
	g.mu.Unlock()
	return nil
}

func (g *Gateway) CurrentConfig() *cliproxyconfig.Config {
	g.mu.RLock()
	defer g.mu.RUnlock()
	return g.current
}

// WaitReload blocks until the service has created its watcher and the reload
// callback has been captured. Run installs it asynchronously, so callers that
// push configuration right after Run would otherwise race it.
func (g *Gateway) WaitReload(ctx context.Context) error {
	ticker := time.NewTicker(10 * time.Millisecond)
	defer ticker.Stop()
	for {
		g.mu.RLock()
		ready := g.reload != nil
		g.mu.RUnlock()
		if ready {
			return nil
		}
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-ticker.C:
		}
	}
}
```
